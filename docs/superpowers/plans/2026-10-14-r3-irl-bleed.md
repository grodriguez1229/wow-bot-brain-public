# R3 IRL-bleed + Output Post-filter Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Cut ambient IRL chatter to ~15% fleet-wide, give every bot a unique persistent IRL identity, kill the shared catchphrase tic, and repair/suppress malformed output lines.

**Architecture:** Three new pure modules — `irl_identity.py` (seeded slot composition), `irl_budget.py` (fleet rolling window + IRL classifier), `postfilter.py` (emoji/length/mash/narration) — wired into `personality.py` (lazy DB migration), `prompts.py` (IRL section gating), and `main.py` (`generate_deduped` pipeline, `/tick`, `/heard`). A tracked `eval/chatter_metrics.py` measures before/after.

**Tech Stack:** Python 3, FastAPI, sqlite3, pytest. No new dependencies.

**Spec:** `docs/superpowers/specs/2026-10-14-r3-irl-bleed-design.md`

## Global Constraints

- Work ONLY in the worktree `~/wow-bot-brain-r3` (branch `r3-irl-bleed`). NEVER edit files in `~/wow-bot-brain` — that checkout is the live hot-reloading brain.
- Python: `~/wow-bot-brain/venv/bin/python` (the worktree has no venv). Run tests from the worktree root: `cd ~/wow-bot-brain-r3 && ~/wow-bot-brain/venv/bin/python -m pytest -q`.
- Tests must never touch the live `memory.db`, MySQL, or Ollama: use the `fresh_db` fixture pattern (monkeypatch `db.DB_PATH`) and monkeypatch gamedb lookups / `call_ollama`.
- Config knobs (exact): `IRL_TARGET=0.15`, `IRL_WINDOW=200`, `SIGNATURE_RATE=0.1`, `TICK_MAX_CHARS=140`, `HEARD_MAX_CHARS=200`, env-overridable with `WBB_` prefix.
- At most 2 LLM calls per generated line.
- Whispers (`/chat`, lane `"default"`): full persona, no IRL budget, no length cap, never silenced.
- The `[tick] <bot> -> <CHANNEL>: "<msg>"` log format must not change (metrics parse it). Silent ticks log `[tick] <bot> -> <CHANNEL>: (silent)` (no quotes).
- `typing_style` strings contain no quote characters (`'` or `"`) — quoted examples get copied verbatim by the 9B model.
- Commit after every task with a message ending in `Co-Authored-By: Claude Opus 5.5 <noreply@anthropic.com>`.

## Review Focus

2. A generated line that is entirely emoji → cleaned to `""` → `/tick` returns an empty message and logs `(silent)`, never a quoted empty line. Test in Task 7.
3. Bots addressing other bots by exact name or possessive ("hey Esterini", "Esterinis sword") must NOT be dropped as mashed names. Test in Task 4.
4. Laughter / elongated words ("hahahahahahaha", "noooooooooooooo") and allowlisted long words ("congratulations") survive the mash filter. Test in Task 4.
5. Cold start after hot-reload (empty window) allows IRL; `IRL_TARGET=0` never allows it. Test in Task 3.

```python
def _seed(bot_name: str) -> int:
    return int.from_bytes(hashlib.sha256(bot_name.encode("utf-8")).digest()[:8], "big")


def _household(rng: random.Random, options: list[str], age: int):
    """-> (text, partner_word | None, terms)."""
    kind = rng.choice(options)
    if kind == "parents":
        return rng.choice(PARENT_HOMES), None, []
    if kind == "alone":
        return "lives alone in a small apartment", None, []
    if kind == "roommates":
        return rng.choice(["lives with a roommate", "has two roommates"]), None, []
    if kind == "dorm":
        return "shares a dorm room with a roommate", None, []
    if kind == "barracks":
        return "bunks in the barracks with their unit", None, []
    female = rng.random() < 0.5
    name = rng.choice(FEMALE_NAMES if female else MALE_NAMES)
    if kind == "partner":
        married = age >= 25 and rng.random() < 0.5
        if married:
            word = "wife" if female else "husband"
        else:
            word = "girlfriend" if female else "boyfriend"
        return f"lives with their {word} {name}", word, [name.lower()]
    # kind == "kids": spouse + 1-3 kids with fixed ages
    word = "wife" if female else "husband"
    n = rng.randint(1, 3)
    kid_ages = sorted(rng.randint(1, min(12, age - 20)) for _ in range(n))
    if n == 1:
        kids = f"one kid (age {kid_ages[0]})"
    else:
        kids = f"{n} kids (ages {', '.join(str(a) for a in kid_ages)})"
    return f"married, {word} is {name}, {kids}", word, [name.lower()]


def _detail(rng: random.Random, cat: str, age: int, foreign: bool):
    """-> (text, terms)."""
    if cat == "pet":
        name = rng.choice(PET_NAMES)
        return f"has a {rng.choice(PET_KINDS)} named {name}", [name.lower()]
    if cat == "hobby":
        return rng.choice(HOBBIES), []
    if cat == "food":
        return rng.choice([t for t, min_age in FOODS if age >= min_age]), []
    if cat == "music":
        return rng.choice(MUSIC), []
    return rng.choice(FOREIGN_REGIONS if foreign else REGIONS), []


def compose_identity(bot_name: str, archetype_id: str | None = None) -> dict:
    """Deterministic person for `bot_name`. Keeps `archetype_id` when it names a known
    archetype (existing bots keep their life stage); otherwise draws one from the seed."""
    seed = _seed(bot_name)
    by_id = {a["id"]: a for a in IRL_IDENTITIES}
    arch = by_id.get(archetype_id or "") or random.Random(seed ^ 0x5EED).choice(IRL_IDENTITIES)
    rng = random.Random(seed)

    age = rng.randint(*arch["ages"])
    occ_text, occ_terms = rng.choice(arch["occupations"])
    house_text, partner_word, house_terms = _household(rng, arch["households"], age)

    details, detail_terms = [], []
    for cat in rng.sample(_DETAIL_CATS, 2):
        text, terms = _detail(rng, cat, age, foreign=arch["id"] == "foreign_player")
        details.append(text)
        detail_terms += terms

    signature, signature_irl = None, False
    if rng.random() < 0.6:
        phrase, signature_irl = rng.choice(arch["signatures"])
        signature = phrase.format(partner=partner_word or "friend")

    facts = [occ_text, house_text] + details
    who_core = f"{age}yo {arch['role']}. {arch['base']}"
    terms = sorted({t.lower() for t in occ_terms + house_terms + detail_terms if t})
    return {
        "archetype": arch["id"],
        "who_core": who_core,
        "irl_facts": facts,
        "irl_terms": terms,
        "signature": signature,
        "signature_irl": bool(signature_irl),
        "typing_style": arch["typing_style"],
        "who": who_core + " " + ". ".join(facts) + ".",
    }
```

```python
"""Fleet-wide IRL budget for ambient chatter (R3) + the shared IRL-line classifier.

A rolling window of the last IRL_WINDOW emitted ambient lines (tick + heard) records
whether each one referenced real life. While the IRL share is below IRL_TARGET the
prompt may offer the bot's real-life details; otherwise it asks for an in-game line.
In-memory, bounded, resets on reload (refills within minutes)."""
from __future__ import annotations

import re
from collections import deque

from config import IRL_TARGET, IRL_WINDOW

_IRL_RE = re.compile("|".join([
    r"\birl\b", r"\breal life\b",
    r"\b(?:at|from|after|before|off|for) work\b", r"\bwork (?:today|tomorrow|tonight|sucked|was)\b",
    r"\bmy (?:job|boss|shift|manager|coworkers?|commute)\b",
    r"\b(?:night|closing|double|morning|day|12.hour) shift\b", r"\bshift (?:today|tomorrow|tonight)\b",
    r"\bcustomers?\b", r"\bclock(?:ed)? (?:in|out)\b",
    r"\b(?:homework|exams?|finals|midterms?|essay|paper due|lecture|professor|teacher|dorm|campus)\b",
    r"\b(?:school|college) (?:today|tomorrow|tonight)\b", r"\bafter school\b",
    r"\bmy (?:gf|bf|girlfriend|boyfriend|wife|husband|partner|kids?|son|daughter|mom|dad|parents|"
    r"roommates?|brother|sister|cat|dog|puppy|kitten)\b",
    r"\b(?:gf|bf|hubby|kiddo|toddler)\b", r"\b(?:wife|husband|mom|dad)\b",
    r"\bkid woke\b", r"\bthe kids\b",
    r"\b(?:beer|coffee|pizza|dinner|lunch|breakfast|takeout|snacks?)\b",
    r"\b(?:go(?:ing)? to bed|need (?:to )?sleep|so tired|exhausted|can'?t sleep|feet are killing)\b",
    r"\b(?:netflix|youtube|spotify|my phone|traffic|rent|bills)\b",
    r"\b(?:barracks|on base|my unit|sergeant|hospital|patients?|scrubs)\b",
]), re.IGNORECASE)

_window: deque = deque(maxlen=IRL_WINDOW)


def is_irl_line(text: str, extra_terms=()) -> bool:
    """True if `text` references the real person's life. `extra_terms` are a bot's own
    distinctive IRL words (pet/partner names, workplace), matched as whole words."""
    text = text or ""
    if _IRL_RE.search(text):
        return True
    return any(re.search(rf"\b{re.escape(t)}\b", text, re.IGNORECASE) for t in extra_terms if t)


def reset_state() -> None:
    _window.clear()


def share() -> float:
    return sum(_window) / len(_window) if _window else 0.0


def irl_allowed() -> bool:
    return share() < IRL_TARGET


def note_ambient_line(text: str, extra_terms=()) -> bool:
    v = is_irl_line(text, extra_terms)
    _window.append(v)
    return v
```

```python
"""Cheap output post-filter for generated chat lines (R3, live-check finding #4). Pure.

clean_line repairs what is mechanical (emoji, mashed bot-name tokens, run-on length);
is_narration flags third-person narration so the caller can re-roll or stay silent."""
from __future__ import annotations

import re

_EMOJI = re.compile(
    "[\U0001F000-\U0001FAFF☀-➿⬀-⯿️‍⃣\U000E0020-\U000E007F]")
_ASTERISK = re.compile(r"\*[^*]{2,}\*")
_LONG_OK = {"congratulations", "responsibility", "transportation", "disappointment",
            "accomplishment", "recommendations", "professionally", "unfortunately",
            "understandable", "characteristics", "internationally", "misunderstanding"}
_NARR_STOP = {"its", "is", "as", "yes", "thanks", "this", "has", "was", "does", "guess",
              "says", "hes", "shes", "thats", "whats", "ups", "hiyas", "heys"}
_MIN_NAME = 5


def has_emoji(text: str) -> bool:
    return bool(_EMOJI.search(text or ""))


def _is_mashed(core: str, names: list[str]) -> bool:
    if core in _LONG_OK or len(set(core)) < 6:          # laughter / elongations
        return False
    for n in names:
        if n in core and len(core) >= len(n) + 3:
            return True
    return len(core) >= 14


def _truncate(text: str, max_len: int) -> str:
    head = text[:max_len]
    cut = max(head.rfind("."), head.rfind("!"), head.rfind("?"))
    if cut >= max_len // 4:
        return head[:cut + 1].strip()
    sp = head.rfind(" ")
    out = head[:sp] if sp > 0 else head
    return out.rstrip(" ,;:-—")


def clean_line(text: str, *, max_len: int | None = None, known_names=()) -> tuple[str, list[str]]:
    flags: list[str] = []
    out = text or ""
    if _EMOJI.search(out):
        out = _EMOJI.sub("", out)
        flags.append("emoji")
    names = [n.lower() for n in known_names if n and len(n) >= _MIN_NAME]
    kept = []
    for tok in out.split():
        core = re.sub(r"[^a-z]", "", tok.lower())
        if core and _is_mashed(core, names):
            if "mashed" not in flags:
                flags.append("mashed")
            continue
        kept.append(tok)
    out = " ".join(kept).strip()
    if max_len and len(out) > max_len:
        out = _truncate(out, max_len)
        flags.append("truncated")
    return out, flags


def is_narration(text: str, bot_name: str) -> bool:
    text = text or ""
    if _ASTERISK.search(text):
        return True
    if not bot_name:
        return False
    n = re.escape(bot_name)
    for m in re.finditer(rf"\b([a-z]+(?:s|es))\s+{n}\b", text, re.IGNORECASE):
        if m.group(1).lower() not in _NARR_STOP:
            return True
    for m in re.finditer(rf"\b{n}\s+([a-z]+(?:s|es))\b", text, re.IGNORECASE):
        if m.group(1).lower() not in _NARR_STOP:
            return True
    return False


def strip_asterisk_actions(text: str) -> str:
    return re.sub(r"\s{2,}", " ", _ASTERISK.sub("", text or "")).strip()
```

```python
async def generate_deduped(gen, *, dedupe: bool, allow_silence: bool = False, decline=None,
                           max_len: int | None = None, bot_name: str = "", known_names=(),
                           silence_on_narration: bool = False):
    """Generate one line via `gen("")`, post-filtered (emoji / mashed names / length).
    `decline` firing -> None. Third-person narration re-rolls once with NARRATION_NUDGE;
    if it persists: None when `silence_on_narration`, else asterisk actions are stripped.
    When `dedupe` is on and the line duplicates a recent fleet line, re-roll once with
    FRESH_NUDGE (only if that stays within 2 LLM calls); a still-duplicate line with
    `allow_silence` -> None. Records the finally-emitted line in the fleet window."""
    calls = 0

    async def _gen(extra):
        nonlocal calls
        calls += 1
        r = await gen(extra)
        if r and r.get("message"):
            cleaned, flags = clean_line(r["message"], max_len=max_len, known_names=known_names)
            if flags:
                print(f"[filter] {bot_name or '?'} {','.join(flags)}")
            r["message"] = cleaned
        return r

    result = await _gen("")
    if decline is not None and decline(result):
        return None
    if result.get("message") and is_narration(result["message"], bot_name):
        print(f"[filter] {bot_name or '?'} narration")
        result = await _gen(NARRATION_NUDGE)
        if decline is not None and decline(result):
            return None
        if result.get("message") and is_narration(result["message"], bot_name):
            if silence_on_narration:
                return None
            result["message"] = strip_asterisk_actions(result["message"])
    if dedupe and result.get("message") and is_duplicate_line(result["message"]):
        if calls < 2:
            result = await _gen(FRESH_NUDGE)
            if allow_silence and result.get("message") and is_duplicate_line(result["message"]):
                return None
        elif allow_silence:
            return None
    if dedupe and result and result.get("message"):
        note_fleet_line(result["message"])
    return result
```

```python
"""Before/after chatter metrics over brain.log-style [tick] lines (R3).

Usage (from repo root):
  venv/bin/python eval/chatter_metrics.py BEFORE.txt AFTER.log [--after MARKER] [--last 1500]
`--after` only counts AFTER-file lines following the last line containing MARKER."""
from __future__ import annotations

import argparse
import collections
import os
import re
import statistics
import sys

sys.path.insert(0, os.path.dirname(os.path.dirname(os.path.abspath(__file__))))
from irl_budget import is_irl_line  # noqa: E402
from postfilter import has_emoji  # noqa: E402

TICK_RE = re.compile(r'^\[tick\] (\S+) -> (\S+): "(.*)"$')
_ROGER = re.compile(r"\b(?:roger|copy)\b", re.IGNORECASE)


def _toks(s: str) -> list[str]:
    return re.findall(r"[a-z']+", s.lower())


def tick_messages(lines, after: str | None = None) -> list[str]:
    lines = list(lines)
    if after:
        idx = max((i for i, l in enumerate(lines) if after in l), default=-1)
        lines = lines[idx + 1:]
    out = []
    for l in lines:
        m = TICK_RE.match(l.strip())
        if m:
            out.append(m.group(3))
    return out


def metrics(msgs: list[str]) -> dict:
    n = len(msgs)
    if not n:
        return {"n": 0}
    sims = []
    sets = [set(_toks(m)) for m in msgs]
    for i, t in enumerate(sets):
        best = 0.0
        for u in sets[max(0, i - 100):i]:
            if t and u:
                best = max(best, len(t & u) / len(t | u))
        sims.append(best)
    tri = collections.Counter()
    for m in msgs:
        w = _toks(m)
        tri.update({" ".join(w[i:i + 3]) for i in range(len(w) - 2)})
    top_tri, top_cnt = tri.most_common(1)[0] if tri else ("", 0)
    openers = collections.Counter(" ".join(_toks(m)[:2]) for m in msgs)
    return {
        "n": n,
        "irl": sum(is_irl_line(m) for m in msgs) / n,
        "roger": sum(bool(_ROGER.search(m)) for m in msgs) / n,
        "top_trigram": top_tri, "top_trigram_share": top_cnt / n,
        "median_len": statistics.median(len(m) for m in msgs),
        "over140": sum(len(m) > 140 for m in msgs) / n,
        "emoji": sum(has_emoji(m) for m in msgs) / n,
        "sim30": sum(s >= .3 for s in sims) / n, "sim40": sum(s >= .4 for s in sims) / n,
        "sim50": sum(s >= .5 for s in sims) / n,
        "openers": [(k, round(v / n, 3)) for k, v in openers.most_common(4)],
    }


def _fmt(label: str, m: dict) -> str:
    if not m.get("n"):
        return f"{label}: n=0"
    return (f"{label}: n={m['n']} IRL {m['irl']:.0%}  roger/copy {m['roger']:.1%}  "
            f"emoji {m['emoji']:.1%}  >140ch {m['over140']:.1%}  median {m['median_len']:.0f}ch  "
            f"top3gram '{m['top_trigram']}' {m['top_trigram_share']:.1%}  "
            f"maxsim>=.3 {m['sim30']:.0%} >=.4 {m['sim40']:.0%} >=.5 {m['sim50']:.0%}  "
            f"openers {m['openers']}")


def main(argv=None) -> None:
    ap = argparse.ArgumentParser()
    ap.add_argument("before")
    ap.add_argument("after_file")
    ap.add_argument("--after", default=None, help="marker line in AFTER file")
    ap.add_argument("--last", type=int, default=1500)
    a = ap.parse_args(argv)
    with open(a.before, errors="ignore") as f:
        before = tick_messages(f)[-a.last:]
    with open(a.after_file, errors="ignore") as f:
        after = tick_messages(f, after=a.after)[-a.last:]
    print(_fmt("Before", metrics(before)))
    print(_fmt("After ", metrics(after)))


if __name__ == "__main__":
    main()
```
